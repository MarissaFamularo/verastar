+// Two separate questions, answered separately and reported separately:
+//   1. Is the QUOTE real evidence?  Exact, unique substring of the supplied corpus.
+//   2. Is the LABEL faithful to it?  Verbatim is accepted as before, bar negation. A non-verbatim label
+//      is accepted only when every word of it is printed in its own quote (it may drop
+//      or reorder source words, never add any), a claim-bearing label keeps the quote's
+//      negation (verbatim or not), and its numbers are typed contextual descriptors printed in
+//      that quote.
+// Neither answer proves the label MEANS what the quote means; the checker judges that
+// against the complete source, and this gate records which labels are paraphrases.
+const LABEL_TOKEN = /[^a-z0-9]+/
+const tokens = s => normalize(s).split(LABEL_TOKEN).filter(Boolean)
+const NEGATION = new Set(['not', 'no', 'never', 'neither', 'nor', 'without', 'lack', 'lacked', 'lacking', 'absence', 'absent'])
+const CLAIM_FIELDS = ['effect', 'qualification']
+function labelFidelity(field, key) {
+  const verbatim = field.quote.includes(field.text)
+  // Compare complete numeric expressions, including adjacent operators, signs,
+  // ranges and units. A substring match cannot prove that 75 is not -75 or >75.
+  const numeric = /(?:[<>≤≥]=?\s*)?[+−-]?(?:\d+(?:[.·]\d+)?|\.\d+)(?:\s*(?:[-–—]|to)\s*[+−-]?\d+(?:[.·]\d+)?)?(?:(?:-|\s+)(?:[a-zA-Z]+)|%|[a-zA-Z]+)?/g
+  const expressions = value => [...value.matchAll(numeric)].map(m => m[0])
+  const sourceExpressions = expressions(field.quote)
+  if (expressions(field.text).some(expression => !sourceExpressions.includes(expression))) return 'numeric-expression-not-in-quote'
+  const quoteTokens = new Set(tokens(field.quote))
+  const labelTokens = tokens(field.text)
+  if (!verbatim && (!labelTokens.length || labelTokens.some(t => !quoteTokens.has(t)))) return 'label-not-in-quote'
+  // A claim-bearing label may not shed its quote's negation, verbatim or not: "differ
+  // significantly" is a substring of "did not differ significantly".
+  const claim = CLAIM_FIELDS.some(k => key.startsWith(k))
+  if (claim && [...quoteTokens].some(t => NEGATION.has(t)) && !labelTokens.some(t => NEGATION.has(t))) return 'label-drops-negation'
+  return verbatim ? 'verbatim' : 'paraphrase'

+// Typed contextual numbers: time units (including prefix and hyphenated forms), short
+// length units for anatomic descriptors, structural analysis counts, and numbered endpoint
+// classifications. No paper-specific values. They are admitted in every field, including
+// qualifications and roles, because "30-day MACEs" is an endpoint name wherever it sits.
+// Anything else numeric stays an explicit coverage failure, never pooled with estimate
+// numbers, and a number a paraphrase carries must be printed in its own quote (fidelity
+// above). The checker must still approve the meaning.
+const UNITS = '(?:seconds?|minutes?|hours?|days?|weeks?|months?|years?|mm|cm)'
+const NUMBER = '(?:[≤≥<>]\\s*)?\\d+(?:\\.\\d+)?(?:\\s*(?:[-–]|to)\\s*\\d+(?:\\.\\d+)?)?'
+const CONTEXTUAL_UNIT = new RegExp(`\\b${UNITS}\\s+${NUMBER}|${NUMBER}(?:-|\\s+)${UNITS}\\b`, 'gi')
+function contextualNumbersStripped(text, key) {

+function locateField(field, source, id, key) {
+  const invalid = reason => ({ invalid: true, reason })
+  if (field == null) return null
+  if (!field || typeof field.text !== 'string' || !field.text.trim() || typeof field.quote !== 'string' || !['text', 'tables'].includes(field.corpus)) return invalid('malformed-context-field')
+  const text = source[field.corpus] || ''
+  const start = text.indexOf(field.quote)
+  if (!field.quote || start < 0) return invalid('context-not-located')
+  if (text.indexOf(field.quote, start + 1) !== -1) return invalid('ambiguous-context-passage')
+  const fidelity = labelFidelity(field, key)
+  if (fidelity !== 'verbatim' && fidelity !== 'paraphrase') return invalid(fidelity)
+  if (!numbersGrounded(contextualNumbersStripped(field.text, key), [])) return invalid('unsupported-context-number')
+  return { text: field.text, fidelity, passageId: `${id}:${key}`, corpus: field.corpus, start, end: start + field.quote.length, quote: field.quote }

+// Preserve the original evidence even when its proposed label cannot render.
+function qualificationLocation(proposed, source) {

+export function prepareResultPlan(handoff, selections) {

+// Strict local validator for the exact JSON Schema subset used by extraction.
+// No defaults, coercion, removal, or repair. Schema extensions fail closed.
+export class ExtractionValidationError extends Error {
+  constructor(diagnostics) {
+    super(`Invalid extraction response: ${diagnostics.join('; ')}`)
+    this.name = 'ExtractionValidationError'
+    this.diagnostics = diagnostics
+    this.retryable = false
+  }
+}
+export function compileExtractionSchema(schema) {
+  const keywords = new Set(['type', 'anyOf', 'properties', 'required', 'additionalProperties', 'items', 'minItems', 'enum', 'description'])
+  const types = new Set(['object', 'array', 'string', 'number', 'integer', 'boolean', 'null'])
+  function inspect(s) {
+    if (!s || typeof s !== 'object' || Array.isArray(s) || Object.keys(s).some(k => !keywords.has(k))) throw new Error('Unsupported extraction schema keyword or node')
+    if (s.anyOf !== undefined) {
+      if (Object.keys(s).some(k => !['anyOf', 'description'].includes(k)) || !Array.isArray(s.anyOf) || !s.anyOf.length) throw new Error('Unsupported extraction schema union')
+      s.anyOf.forEach(inspect); return
+    }
+    if (!types.has(s.type)) throw new Error('Unsupported extraction schema type')
+    const appropriate = { properties: 'object', required: 'object', additionalProperties: 'object', items: 'array', minItems: 'array' }
+    if (Object.entries(appropriate).some(([key, type]) => key in s && s.type !== type)) throw new Error('Misplaced extraction schema keyword')
+    if (s.enum !== undefined && (!Array.isArray(s.enum) || !s.enum.length || s.enum.some(x => x !== null && !['string', 'number', 'boolean'].includes(typeof x)))) throw new Error('Unsupported extraction schema enum')
+    if (s.type === 'object') {
+      if (!s.properties || typeof s.properties !== 'object' || Array.isArray(s.properties) || s.additionalProperties !== false || !Array.isArray(s.required)
+        || s.required.length !== Object.keys(s.properties).length || new Set(s.required).size !== s.required.length || s.required.some(k => !Object.hasOwn(s.properties, k))) throw new Error('Unsupported extraction object schema')
+      Object.values(s.properties).forEach(inspect)
+    }
+    if (s.type === 'array') {
+      if (s.minItems !== undefined && (!Number.isSafeInteger(s.minItems) || s.minItems < 0)) throw new Error('Unsupported extraction array schema')
+      inspect(s.items)
+    }
+  }
+  inspect(schema)
+  function check(s, v, path) {
+    if (s.anyOf) return s.anyOf.some(branch => check(branch, v, path).length === 0) ? [] : [`${path}: no union alternative matches`]
+    const matches = s.type === 'null' ? v === null : s.type === 'array' ? Array.isArray(v)
+      : s.type === 'object' ? v !== null && typeof v === 'object' && !Array.isArray(v)
+        : s.type === 'number' ? typeof v === 'number' && Number.isFinite(v)
+          : s.type === 'integer' ? Number.isSafeInteger(v) : typeof v === s.type
+    if (!matches) return [`${path}: expected ${s.type}`]
+    const errors = []
+    if (s.enum && !s.enum.includes(v)) errors.push(`${path}: invalid enum`)
+    if (s.type === 'object') {
+      for (const key of s.required) if (!Object.hasOwn(v, key)) errors.push(`${path}.${key}: missing required field`)
+      for (const key of Object.keys(v)) {
+        if (!Object.hasOwn(s.properties, key)) errors.push(`${path}.${key}: unexpected field`)
+        else errors.push(...check(s.properties[key], v[key], `${path}.${key}`))
+      }
+    }
+    if (s.type === 'array') {
+      if (v.length < (s.minItems || 0)) errors.push(`${path}: too few items`)
+      for (let i = 0; i < v.length; i++) errors.push(...check(s.items, v[i], `${path}[${i}]`))
+    }
+    return errors
+  }
+  return value => {
+    const errors = check(schema, value, '$')
+    if (errors.length) throw new ExtractionValidationError(errors)
+    return value
+  }
+}
+
+export function validateExtractionSchema(schema, value) {
+  return compileExtractionSchema(schema)(value)
